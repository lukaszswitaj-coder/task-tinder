async function loadTasks() { const card = document.getElementById('card'); const taskTitle = document.getElementById('task-title'); const taskListName = document.getElementById('task-list-name'); taskTitle.innerText = "Ładowanie zadań..."; card.style.display = 'flex'; try { const listsRes = await fetch('https://tasks.googleapis.com/tasks/v1/users/@me/lists', { headers: { Authorization: `Bearer ${accessToken}` } }); const listsData = await listsRes.json(); const taskLists = listsData.items || []; tasks = []; for (const list of taskLists) { const items = await fetchAllTasksFromList(list.id); // BRAK FILTROWANIA - bierzemy WSZYSTKIE nieukończone zadania const mapped = items .filter(t => t.title && t.title.trim() !== '') .map(t => ({ ...t, listId: list.id, listTitle: list.title })); tasks = tasks.concat(mapped); } if (tasks.length === 0) { taskTitle.innerText = "API zwróciło 0 zadań na tej liście!"; } else { // Wyświetlamy tytuł zadania ORAZ to, co dokładnie przychodzi w polu `due` const first = tasks[0]; const todayISO = new Date().toISOString(); taskListName.innerText = first.listTitle; taskTitle.innerHTML = ` Tytuł: ${first.title}

Data w API (due): ${first.due || 'brak daty'}
Dzisiaj (ISO): ${todayISO} `; } } catch (err) { console.error(err); taskTitle.innerText = "Błąd połączenia z API!"; } }
